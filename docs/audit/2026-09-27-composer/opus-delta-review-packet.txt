105:     ? { ...state, finals: [...state.finals, event.text], interim: "" }
106:     : { ...state, interim: event.text };
107: }
108:
109: function failed(state: DictationState, code: string): DictationState {
110:   if (IGNORED_ERRORS.includes(code)) return state;
111:   // An error the spec names copy for is theirs to read and fix; anything else — including an error
112:   // nobody planned for — is a reason to use the server instead, silently.
113:   if (PERMISSION_ERRORS.includes(code)) {
114:     return { ...initialDictation(state.fallback), phase: "error",
115:       notice: { text: PERMISSION_COPY, ms: 0 } };
116:   }
117:   if (code === "no-speech" || code === MAX_DURATION_ERROR) {
118:     // Safari raises `no-speech` after a pause, mid-hold, with sentences already final. Those are
119:     // their words: they finalise normally, and only a hold that heard NOTHING gets the line
120:     // (panel round 1, HIGH 5).
121:     if (transcriptOf(state).length > 0) return settle(state);
122:     return { ...initialDictation(state.fallback), phase: "error",
123:       notice: { text: NO_SPEECH_COPY, ms: 0 } };
124:   }
125:   return initialDictation(true);
126: }
127:
128: export function nextDictation(state: DictationState, event: DictationEvent): DictationState {
129:   switch (event.type) {
130:     case "press":
131:       return { ...initialDictation(state.fallback), phase: "listening", pressedAt: event.at };
132:     case "segment": {
133:       if (state.phase !== "listening" && state.phase !== "finalising") return state;
134:       const next = heard(state, event);
135:       // The last final after a release is what the wait in `finalising` was for.
136:       return state.phase === "finalising" && event.final ? settle(next) : next;
137:     }
138:     case "release": {
139:       if (state.phase !== "listening") return state;
140:       if (state.pressedAt !== null && event.at - state.pressedAt < TAP_MS) {
141:         return { ...initialDictation(state.fallback), notice: { text: HOLD_HINT, ms: HINT_MS } };
142:       }
143:       // Nothing is pending when every segment has been finalised, so there is no reason to spend
144:       // the finalise timer waiting for an API that has already said everything it is going to.
145:       return state.interim === "" && state.finals.length > 0
146:         ? settle(state)
147:         : { ...state, phase: "finalising" };
148:     }
149:     case "finalised":
150:       return state.phase === "finalising" ? settle(state) : state;
151:     case "error":
152:       return failed(state, event.code);
153:     case "fallback": {
154:       // Whatever the user said still counts; a recogniser that keeps ending is a capability problem,
155:       // and §3 answers those silently rather than with a line the user cannot act on.
156:       const settled = settle({ ...state, fallback: true });
157:       return settled.phase === "empty" ? initialDictation(true) : settled;
158:     }
159:     case "settled":
160:       return { ...initialDictation(state.fallback), notice: state.notice };
161:     case "notice_expired":
162:       return { ...state, notice: null };
163:   }
164: }

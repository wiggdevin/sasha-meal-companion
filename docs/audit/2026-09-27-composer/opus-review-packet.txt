26:       setEditing(true);

28:     onFocus: () => setEditing(true),

236:     if (handle === null) return;
237:     if (!this.holding) {

239:       handle.cancel();

278:   private async consumeClip(clip: Blob): Promise<void> {

280:     this.recorder = null;

289:     if (gen !== this.gen) return;

292:     if (!outcome.ok) { this.abandon({ text: outcome.message, ms: 0 }); return; }

295:     this.closeDictation();

297:     this.ports.writeBox(this.withDraft(text));

299:     this.showNotice(null);

302:   }

306:   private settleSpeech(next: DictationState): void {
307:     this.clearTimer(this.finaliseTimer);

313:     this.gen += 1;

318:   }

320:   /**
321:    * Ruling R1: a hold does not throw away what the user had already typed. Half a sentence in the box

323:    * exactly the transcript, which is every hold that starts from an empty composer.
324:    */

326:     return [this.typed.trim(), this.carried.trim(), text.trim()]
327:       .filter((part) => part.length > 0).join(" ");
328:   }

332:     this.gen += 1;
333:     this.ports.writeBox(this.typed);

336:     this.apply({ type: "settled" });
337:     this.publish();
338:   }
339:
340:   /** Nothing happened and nothing is said about it. */
341:   private quiet(): void {
342:     this.abandon(null);
343:   }
344:
345:   /** §4's budget: release → the transcript. It joins the run the turn's own `sent` opens. */
346:   private closeDictation(): void {
347:     markMs("coach-turn", "dictation", Math.max(0, this.now() - this.releasedAt));
348:   }

354:     return this.state;
355:   }

364:         : notice?.text ?? null;

367:       this.noticeTimer = null;
368:       this.notice = null;
369:       this.apply({ type: "notice_expired" });

371:     }, notice.ms);

375:     this.ports.publish(this.view());
376:   }

384:     // second click is always deliberate, including a fast Stop after permission is denied.

93: /** The end of a hold: what was heard becomes a turn, or the empty line. */
94: function settle(state: DictationState): DictationState {
95:   const text = transcriptOf(state);
96:   const cleared = { ...state, finals: [], interim: "" };
97:   if (text.length === 0) {
98:     return { ...cleared, phase: "empty", text: "", notice: { text: EMPTY_COPY, ms: EMPTY_MS } };
99:   }

109: function failed(state: DictationState, code: string): DictationState {
110:   if (IGNORED_ERRORS.includes(code)) return state;
111:   // An error the spec names copy for is theirs to read and fix; anything else — including an error
112:   // nobody planned for — is a reason to use the server instead, silently.
113:   if (PERMISSION_ERRORS.includes(code)) {
114:     return { ...initialDictation(state.fallback), phase: "error",
115:       notice: { text: PERMISSION_COPY, ms: 0 } };
116:   }
117:   if (code === "no-speech" || code === MAX_DURATION_ERROR) {
118:     // Safari raises `no-speech` after a pause, mid-hold, with sentences already final. Those are
119:     // their words: they finalise normally, and only a hold that heard NOTHING gets the line
120:     // (panel round 1, HIGH 5).
121:     if (transcriptOf(state).length > 0) return settle(state);
122:     return { ...initialDictation(state.fallback), phase: "error",
123:       notice: { text: NO_SPEECH_COPY, ms: 0 } };
124:   }
125:   return initialDictation(true);
126: }

128: export function nextDictation(state: DictationState, event: DictationEvent): DictationState {
129:   switch (event.type) {
130:     case "press":
131:       return { ...initialDictation(state.fallback), phase: "listening", pressedAt: event.at };
132:     case "segment": {
133:       if (state.phase !== "listening" && state.phase !== "finalising") return state;
134:       const next = heard(state, event);
135:       // The last final after a release is what the wait in `finalising` was for.
136:       return state.phase === "finalising" && event.final ? settle(next) : next;
137:     }
138:     case "release": {
139:       if (state.phase !== "listening") return state;
140:       if (state.pressedAt !== null && event.at - state.pressedAt < TAP_MS) {
141:         return { ...initialDictation(state.fallback), notice: { text: HOLD_HINT, ms: HINT_MS } };
142:       }
143:       // Nothing is pending when every segment has been finalised, so there is no reason to spend
144:       // the finalise timer waiting for an API that has already said everything it is going to.
145:       return state.interim === "" && state.finals.length > 0
146:         ? settle(state)
147:         : { ...state, phase: "finalising" };
148:     }
149:     case "finalised":
150:       return state.phase === "finalising" ? settle(state) : state;
151:     case "error":
152:       return failed(state, event.code);
153:     case "fallback": {
154:       // Whatever the user said still counts; a recogniser that keeps ending is a capability problem,
155:       // and §3 answers those silently rather than with a line the user cannot act on.
156:       const settled = settle({ ...state, fallback: true });
157:       return settled.phase === "empty" ? initialDictation(true) : settled;
158:     }
159:     case "settled":
160:       return { ...initialDictation(state.fallback), notice: state.notice };
161:     case "notice_expired":
162:       return { ...state, notice: null };
163:   }

118:   recognition.interimResults = true;

120:   recognition.maxAlternatives = 1;

131:   recognition.onerror = (event) => { if (live) handlers.onError(event.error ?? "unknown"); };
132:   recognition.onend = () => {
133:     if (!live) return;
134:     live = false;
135:     handlers.onEnd();
136:   };
137:   let opened = false;
138:   const audioStarted = (): void => {
139:     if (!live || opened) return;
140:     opened = true;
141:     handlers.onAudioStart?.();
142:   };
143:   recognition.onaudiostart = audioStarted;
144:   recognition.onstart = audioStarted;
145:   try {
146:     recognition.start();
147:   } catch (e) {
148:     // Safari throws InvalidStateError when a recogniser is started twice, and a throw here would
149:     // otherwise leave the caller holding a session that never ends (panel round 1, LOW 22).
150:     live = false;
151:     handlers.onError(e instanceof Error && e.name ? e.name : "start-failed");

153:   }

160:    */
161:   return {

13:  *    release. So this session keeps ONE running transcript (finalised segments plus the current

1: /**
2:  * The microphone, once, for both surfaces that hold a button to record.
3:  *
4:  * `components/VoiceCapture.tsx` (the FAB's voice note) and the coach's voice fallback
5:  * (`lib/coach-transcribe-client.ts`, T7 §2) open the same microphone under the same rules: the
6:  * bucket's own mime types, a 30 s hard stop, and the two failures the user is told apart. This module
7:  * is where that lives, so a `lib/` module never has to import a component to get at it.
8:  *
9:  * TIMESLICE: `MediaRecorder.start()` with no argument buffers the whole clip and, on iOS, has
10:  * been observed to deliver an EMPTY blob when the recorder is stopped quickly. Starting with a
11:  * timeslice makes the recorder hand over a chunk every quarter second, so a short hold still ends
12:  * with bytes in it.
13:  */
14:
15: import { pickAudioMimeType } from "@/lib/capture/presentation";
16:
17: /** The two ways opening the microphone can fail — kept distinct so the user is told which one it was
18:  *  (audit #73): a denied permission is theirs to fix in Settings, a true device/codec gap is not. */
19: export type MicFailureReason = "unsupported" | "permission-denied";
20:
21: /**
22:  * `getUserMedia` rejects with a `DOMException` named `NotAllowedError` when the prompt is
23:  * dismissed or the permission was already denied; every other rejection (no device, `NotFoundError`,
24:  * etc.) is a real capability gap. Duck-typed on `.name` rather than `instanceof DOMException` so a
25:  * plain object with that name (as a test, or a non-browser environment, would produce) still
26:  * classifies correctly.
27:  */
28: export function classifyMicError(error: unknown): MicFailureReason {
29:   const name = (error as { name?: unknown } | null | undefined)?.name;
30:   return name === "NotAllowedError" ? "permission-denied" : "unsupported";
31: }
32:
33: const UNSUPPORTED_COPY: Record<MicFailureReason, string> = {
34:   unsupported: "This phone will not record here. Type it instead, or use search.",
35:   "permission-denied": "Microphone access is off. Allow it in Settings, then try again.",
36: };
37:
38: export function unsupportedCopy(reason: MicFailureReason): string {
39:   return UNSUPPORTED_COPY[reason];
40: }

42: /** The hard stop. The server refuses a longer clip, and the countdown is the only warning. */
43: export const MAX_RECORDING_SECONDS = 30;
44: /** How often the recorder hands over what it has. See TIMESLICE above. */
45: export const RECORDER_TIMESLICE_MS = 250;
46: const TICK_MS = 100;
47:
48: export type MicRecorder = {
49:   /** End the clip and deliver it through `onClip`. */
50:   stop: () => void;
51:   /** End the clip and throw it away — a hold that turned out not to want it. */
52:   cancel: () => void;
53:   mimeType: string;
54: };
55:
56: export type MicHandlers = {
57:   onClip: (clip: Blob) => void;
58:   onFailure: (reason: MicFailureReason, copy: string) => void;
59:   onElapsed?: (seconds: number) => void;
60: };
61:
62: type MicScope = {
63:   MediaRecorder?: {
64:     new (stream: MediaStream, options: { mimeType: string }): MediaRecorder;
65:     isTypeSupported: (type: string) => boolean;
66:   };
67:   navigator?: { mediaDevices?: { getUserMedia: (c: MediaStreamConstraints) => Promise<MediaStream> } };
68:   setInterval?: typeof setInterval;
69:   clearInterval?: typeof clearInterval;
70: };
71:
72: /**
73:  * Open the microphone and start recording, or report why not. The caller holds the handle: `stop`
74:  * delivers the clip, `cancel` drops it, and the 30 s ceiling calls `stop` itself.
75:  */
76: export async function openMicRecorder(
77:   handlers: MicHandlers, scope: MicScope = globalThis as MicScope,
78: ): Promise<MicRecorder | null> {
79:   const Recorder = scope.MediaRecorder;
80:   const mimeType = pickAudioMimeType(
81:     (type) => Recorder !== undefined && Recorder.isTypeSupported(type),
82:   );
83:   const devices = scope.navigator?.mediaDevices;
84:   if (Recorder === undefined || mimeType === null || devices === undefined) {
85:     handlers.onFailure("unsupported", unsupportedCopy("unsupported"));
86:     return null;
87:   }
88:
89:   let stream: MediaStream;
90:   try {
91:     stream = await devices.getUserMedia({ audio: true });
92:   } catch (error) {
93:     const reason = classifyMicError(error);
94:     handlers.onFailure(reason, unsupportedCopy(reason));
95:     return null;
96:   }
97:
98:   const chunks: Blob[] = [];
99:   const media = new Recorder(stream, { mimeType });
100:   let wanted = true;
101:   let timer: ReturnType<typeof setInterval> | null = null;
102:   const clearTick = (): void => {
103:     if (timer !== null) (scope.clearInterval ?? clearInterval)(timer);
104:     timer = null;
105:   };
106:   media.ondataavailable = (event) => { if (event.data.size > 0) chunks.push(event.data); };
107:   media.onstop = () => {
108:     clearTick();
109:     for (const track of stream.getTracks()) track.stop();
110:     if (wanted) handlers.onClip(new Blob(chunks, { type: mimeType }));
111:   };
112:   media.start(RECORDER_TIMESLICE_MS);
113:
114:   const end = (): void => {
115:     clearTick();
116:     if (media.state === "recording") media.stop();
117:   };
118:   const startedAt = Date.now();
119:   timer = (scope.setInterval ?? setInterval)(() => {
120:     const seconds = (Date.now() - startedAt) / 1000;
121:     handlers.onElapsed?.(seconds);
122:     if (seconds >= MAX_RECORDING_SECONDS) end();
123:   }, TICK_MS);
124:   return {
125:     mimeType,
126:     stop: end,
127:     cancel: () => { wanted = false; end(); },
128:   };
129: }

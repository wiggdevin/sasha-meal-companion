5: export function useComposerInput(disabled: boolean, onSend: (text: string) => void | boolean) {
6:   const input = useRef<HTMLTextAreaElement>(null);
7:   const composing = useRef(false);
8:   const recordPointerDown = useRef(false);
9:   const cancelledPointerClick = useRef(false);
10:   const wasListening = useRef(false);
11:   const [draft, setDraft] = useState("");
12:   const [editing, setEditing] = useState(false);
13:   const [listening, setListening] = useState(false);
14:   const [reviewNeeded, setReviewNeeded] = useState(false);
15:   const [micNotice, setMicNotice] = useState<string | null>(null);
16:
17:   const send = (text: string) => {
18:     if (!text.trim() || disabled || listening || reviewNeeded) return;
19:     if (onSend(text) !== false && input.current) {
20:       input.current.value = "";
21:       setDraft("");
22:       setEditing(false);
23:     }
24:   };
25:   const readBox = () => input.current?.value ?? "";
26:   const textEvents = {
27:     onChange: (event: { currentTarget: HTMLTextAreaElement }) => {
28:       setDraft(event.currentTarget.value);
29:       setEditing(true);
30:     },
31:     onFocus: () => setEditing(true),
32:     onClick: () => setEditing(true),
33:     onSelect: () => setEditing(true),
34:     onBlur: (event: FocusEvent<HTMLTextAreaElement>) => {
35:       const toAction = (event.relatedTarget as HTMLElement | null)
36:         ?.closest?.("[data-composer-action]");
37:       if (!recordPointerDown.current && !toAction) setEditing(false);
38:     },
39:     onCompositionStart: () => { composing.current = true; },
40:     onCompositionEnd: () => { composing.current = false; },
41:     onKeyDown: (event: KeyboardEvent<HTMLTextAreaElement>) => {
42:       if (event.key === "Enter" && !event.shiftKey && !composing.current
43:         && !event.nativeEvent.isComposing && event.keyCode !== 229) {
44:         event.preventDefault();
45:         send(event.currentTarget.value);
46:       }
47:     },
48:   };
49:   const micEvents = {
50:     readBox,
51:     writeBox: (text: string) => {
52:       if (input.current) input.current.value = text;
53:       setDraft(text);
54:     },
55:     onSend: send,
56:     reviewNeeded,
57:     onReviewNeeded: (needed: boolean) => {
58:       setReviewNeeded(needed);
59:       if (!needed) setEditing(false);
60:     },
61:     readyToSend: Boolean(draft.trim()) && !editing,
62:     onRecordPointerDown: () => {
63:       cancelledPointerClick.current = false;
64:       recordPointerDown.current = true;
65:     },
66:     onRecordPointerDone: () => {
67:       recordPointerDown.current = false;
68:       cancelledPointerClick.current = true;
69:     },
70:     onActionStart: () => { cancelledPointerClick.current = false; },
71:     onActionClick: (detail: number) => {
72:       recordPointerDown.current = false;
73:       const cancelled = cancelledPointerClick.current;
74:       cancelledPointerClick.current = false;
75:       return cancelled && detail > 0;
76:     },
77:     onActionPointerUp: (event: PointerEvent<HTMLElement>) => {
78:       const box = event.currentTarget.getBoundingClientRect();
79:       const outside = event.clientX < box.left || event.clientX > box.right
80:         || event.clientY < box.top || event.clientY > box.bottom;
81:       if (outside) {
82:         recordPointerDown.current = false;
83:         cancelledPointerClick.current = true;
84:       }
85:       return outside;
86:     },
87:     onListening: (locked: boolean) => {
88:       setListening(locked);
89:       if (wasListening.current && !locked) setEditing(false);
90:       wasListening.current = locked;
91:     },
92:     onNotice: setMicNotice,
93:   };
94:   return { input, draft, editing, listening, reviewNeeded, micNotice, readBox, send, textEvents, micEvents };
